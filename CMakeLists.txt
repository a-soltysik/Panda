cmake_minimum_required(VERSION 3.28)

include(cmake/Options.cmake)
if(PANDA_DOCS_ONLY)
    project(PandaDocumentation LANGUAGES NONE)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    include(cmake/DocumentationChecks.cmake)
    return()
endif()

include(cmake/MsvcEnvironment.cmake)
project(Panda VERSION 0.1.0 LANGUAGES CXX)
if(PROJECT_IS_TOP_LEVEL)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
endif()

if(NOT PANDA_BUILD_ENGINE AND (PANDA_BUILD_SCENE OR PANDA_BUILD_TOOLS OR PANDA_BUILD_CUDA OR
        PANDA_BUILD_EXAMPLES OR PANDA_BUILD_BENCHMARKS))
    message(FATAL_ERROR "PANDA_BUILD_SCENE, PANDA_BUILD_TOOLS, PANDA_BUILD_CUDA, PANDA_BUILD_EXAMPLES and PANDA_BUILD_BENCHMARKS require PANDA_BUILD_ENGINE=ON")
endif()
if(PANDA_BUILD_BENCHMARKS AND NOT PANDA_BUILD_TOOLS)
    message(FATAL_ERROR "PANDA_BUILD_BENCHMARKS requires PANDA_BUILD_TOOLS=ON for panda-bench")
endif()
include(cmake/CompilerWarnings.cmake)
include(cmake/Coverage.cmake)
include(cmake/Sanitizers.cmake)
include(cmake/Stacktrace.cmake)
include(cmake/Dependencies.cmake)
if(PANDA_BUILD_ENGINE)
    find_package(Vulkan 1.3 REQUIRED)
endif()
if(PANDA_BUILD_CUDA)
    find_package(CUDAToolkit REQUIRED)
endif()
include(cmake/StaticAnalysis.cmake)
include(cmake/Formatting.cmake)
include(cmake/Targets.cmake)
include(cmake/BuiltinShaders.cmake)
include(cmake/ShaderResources.cmake)

if(PANDA_BUILD_TESTS)
    enable_testing()
endif()

add_subdirectory(src)

if(PANDA_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()

if(PANDA_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()

if(PANDA_BUILD_TESTS)
    add_subdirectory(tests)
endif()

if(PANDA_BUILD_DOCS)
    add_subdirectory(docs/api)
endif()

if(PANDA_BUILD_DOC_CHECKS)
    include(cmake/DocumentationChecks.cmake)
endif()

panda_finalize_format_checks()
